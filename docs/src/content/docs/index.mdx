import { Card, CardGrid } from '@astrojs/starlight/components'

```bash
composer require silarhi/cursor-pagination
```

<CardGrid>
    <Card title="Keyset pagination" icon="rocket">
        Each page is fetched with a `WHERE` clause built from the last row of the previous page, never an `OFFSET`.
    </Card>
    <Card title="Multi-field ordering" icon="list-format">
        Combine several fields, each ascending or descending, ending with a unique one.
    </Card>
    <Card title="Item or chunk iteration" icon="puzzle">
        Iterate entity by entity, or by arrays of `maxPerPages` items for flush and clear batches.
    </Card>
    <Card title="Restartable iterations" icon="approve-check">
        Every iteration starts from the beginning with its own cursor, even if a previous one was stopped early.
    </Card>
    <Card title="Fetch joins handled" icon="setting">
        One query per page, or two when the query joins a collection: the page's ids, then their rows.
    </Card>
    <Card title="Query analysis" icon="warning">
        Collection joins and scalar rows are detected; misconfigured orders throw instead of skipping rows.
    </Card>
    <Card title="Counting" icon="document">
        `count()` and `getNbPages()`, with a plain `COUNT` when no join can duplicate rows.
    </Card>
</CardGrid>
